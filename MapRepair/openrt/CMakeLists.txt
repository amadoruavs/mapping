cmake_minimum_required(VERSION 3.16)
project(MapRepairOpenRT LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs highgui features2d ml)
set(OPENRT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/OpenRT")
set(OPENRT_VERSION_MAJOR 1)
set(OPENRT_VERSION_MINOR 0)
set(OPENRT_VERSION_PATCH 0)
set(MAX_RAY_COUNTER 2)
set(ENABLE_BSP ON)
set(ENABLE_PDP ON)
set(ENABLE_CACHE OFF)
set(SHADOW_TYPE FAST)
configure_file("${OPENRT_ROOT}/cmake/types.h.in" "${CMAKE_CURRENT_BINARY_DIR}/generated/types.h")
file(GLOB OPENRT_SOURCES "${OPENRT_ROOT}/modules/core/*.cpp")
add_library(openrt_core STATIC ${OPENRT_SOURCES})
target_include_directories(openrt_core PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/generated"
    "${OPENRT_ROOT}/include" "${OPENRT_ROOT}/modules" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(openrt_core PUBLIC ${OpenCV_LIBS})
add_executable(maprepair_openrt trace.cpp)
target_link_libraries(maprepair_openrt PRIVATE openrt_core)
if(MSVC)
  target_compile_options(openrt_core PRIVATE /O2)
else()
  target_compile_options(openrt_core PRIVATE -O3)
endif()
