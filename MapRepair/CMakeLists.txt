cmake_minimum_required(VERSION 3.16)
project(MapRepair LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(maprepair SHARED native.cpp)
target_compile_features(maprepair PRIVATE cxx_std_17)
target_link_libraries(maprepair PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(maprepair PRIVATE /O2 /fp:precise)
else()
  target_compile_options(maprepair PRIVATE -O3 -ffp-contract=off)
endif()
